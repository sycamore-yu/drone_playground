"""Prepare explicit selection and durable copies; never change experiment sources."""
import json
from pathlib import Path
import shutil

ROOT = Path(__file__).resolve().parents[2]
DEST = ROOT/'docs/verification/final-acceptance'
DEST.mkdir(parents=True, exist_ok=True)
cells=[]
for method in ('ppo','bptt','shac','pointcloud','super','ego_planner'):
    for task in ('hovering','tracking','racing','navigation/static','navigation/dynamic'):
        item=dict(method=method, task=task)
        if task.startswith('navigation/'):
            family=task.split('/')[-1]
            if method in ('super','ego_planner'):
                item['run_id']=f'final-acceptance-native-{method}-{family}-v2'
            elif method=='pointcloud':
                item.update(run_id='final-acceptance-pointcloud-navigation-30000-v2',
                            scene_prefix='S' if family=='static' else 'D',
                            scope='Original paper reconstruction checkpoint at 30000/50000 updates; 4 speeds per fixed scene')
            else:
                version='v2' if method=='ppo' else 'v3'
                item.update(run_id=f'final-acceptance-heldout-{method}-navigation-{family}-v2',
                            training_run=f'final-acceptance-{method}-navigation-{family}-smoke-{version}',
                            scope='4096-interaction engineering trial; navigation convergence optional')
        elif method in ('super','ego_planner'):
            version='v3' if task=='racing' else 'v2'
            item.update(run_id=f'final-acceptance-native-{method}-{task}-{version}',
                        scope='Explicit time-reference to native receding goals; racing starts with counted vertical takeoff fallback')
        else:
            version='t2' if method=='pointcloud' and task=='hovering' else 't1' if method=='pointcloud' or (method=='shac' and task=='racing') else 't0'
            suffix='trained-v1' if method=='shac' and task=='racing' else 'v1'
            item.update(run_id=f'final-acceptance-heldout-{method}-{task}-{suffix}',
                        training_run=f'final-acceptance-{method}-{task}-{version}',
                        scope='Platform point-cloud control transfer with full encoder and explicit input scale 0.02' if method=='pointcloud' else 'Canonical control task, 25–50ms randomized delay')
        cells.append(item)
(DEST/'selection.json').write_text(json.dumps(dict(cells=cells),indent=2)+'\n')
for run_id in sorted({v[key] for v in cells for key in ('run_id','training_run') if key in v}):
    source=ROOT/'experiments'/run_id
    destination=DEST/'runs'/run_id
    destination.mkdir(parents=True,exist_ok=True)
    for relative in ['eval/report.json','result.json','resolved-config.json','manifest.json','command.txt']:
        src=source/relative
        if src.exists():
            out=destination/relative
            out.parent.mkdir(parents=True,exist_ok=True)
            shutil.copyfile(src,out)
for directory in (ROOT/'tmp/final-acceptance/trials').glob('*'):
    for name in ('observation.json','manifest.json','proposal.json'):
        source=directory/name
        if source.exists():
            target=DEST/'tuning'/directory.name/name
            target.parent.mkdir(parents=True,exist_ok=True)
            shutil.copyfile(source,target)
for name in ['legacy-checkpoints.json','pointcloud-feature-diagnosis.json',
             'shac-trained-endpoint-selection.json','remaining-frozen-evaluations.json',
             'native-control-exits-v2.json']:
    source=ROOT/'tmp/final-acceptance'/name
    if source.exists():
        shutil.copyfile(source,DEST/name)
print('selection cells',len(cells),'archived sources',len(list((DEST/'runs').iterdir())))
