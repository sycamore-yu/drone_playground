def command(source):
    pieces=shlex.split(source)
    entry=pieces[1] if len(pieces)>1 else ''
    if entry in ('scripts/train.py','scripts/eval.py'):
        pieces=['pixi','run',Path(entry).stem,*pieces[2:]]
    return shlex.join([piece.replace(str(ROOT)+'/', '') for piece in pieces])
