"""Read each selected replay, verify terminal length, and exercise play once per method."""
import hashlib
import json
import os
from pathlib import Path
import pickle
import subprocess
import sys

import numpy as np

ROOT=Path(__file__).resolve().parents[2]
OUTPUT=ROOT/'docs/verification/final-acceptance'
selection=json.loads((OUTPUT/'selection.json').read_text())['cells']
checked=[]
for run_id in sorted({row['run_id'] for row in selection}):
    run=ROOT/'experiments'/run_id
    report=json.loads((run/'eval/report.json').read_text())
    indices={}
    for index in (run/'rollouts').rglob('episodes.json'):
        for episode in json.loads(index.read_text()):
            indices[(index.parent/episode['file']).resolve()]=episode
    for file in sorted((run/'rollouts').rglob('*.mj_unroll')):
        before=hashlib.sha256(file.read_bytes()).hexdigest()
        with file.open('rb') as f:
            record=pickle.load(f)
        # The installed RScope writer serializes its Rollout NamedTuple.
        trace=record._asdict()
        if not isinstance(trace,dict) or 'time' not in trace:
            raise ValueError(f'Unknown replay payload {file}: {type(record)}')
        times=np.asarray(trace['time'])
        if times.ndim==1:
            times=times[:,None]
        if not np.isfinite(times).all() or (np.diff(times,axis=0)<=0).any():
            raise ValueError(f'Invalid or post-terminal padded clock {file}')
        expected=indices.get(file.resolve())
        if expected:
            assert len(times)==expected['frames']
            np.testing.assert_allclose(times[-1,0],expected['final_time_s'],atol=2e-5)
        relative=file.relative_to(run/'rollouts')
        parts=relative.parts
        if 'cells' in report and not (report.get('episodes') and 'scene_id' in report['episodes'][0]):
            difficulty=parts[0]
            case=int(parts[1].split('-')[-1])
            terminal=report['cells'][difficulty]['episodes'][case]
        elif report['episodes'] and 'scene_id' in report['episodes'][0]:
            speed=float(parts[0].removeprefix('speed-'))
            terminal=next(row for row in report['episodes']
                          if row['scene_id']==parts[1] and row['command_speed_m_s']==speed)
        else:
            case=int(parts[0].split('-')[-1]) if parts[0].startswith('case-') else 0
            if len(list((run/'rollouts').rglob('*.mj_unroll'))) < report['num_trials']:
                worst=max(report['episodes'],key=lambda row:(row['failed'],row['rmse_m']))['case']
                case=sorted(set(range(min(4,report['num_trials'])))|{worst})[case]
            terminal=report['episodes'][case]
        proof_path=file.parent/'readback-verification.json'
        proof=json.loads(proof_path.read_text()) if proof_path.exists() else {}
        initial=bool(proof.get('initial_frame_included',False))
        assert len(times)==terminal['steps']+int(initial),(file,len(times),terminal['steps'],initial)
        if initial:
            assert times[0,0]==0
        terminal_time=terminal.get('elapsed_s',terminal.get('duration_s',terminal['steps']*.02))
        np.testing.assert_allclose(times[-1,0],terminal_time,atol=2e-4)
        for name in ('qpos','qvel','mocap_pos','mocap_quat'):
            if name in trace:
                assert np.isfinite(np.asarray(trace[name])).all(),(file,name)
        assert hashlib.sha256(file.read_bytes()).hexdigest()==before
        checked.append(dict(file=str(file.relative_to(ROOT)),sha256=before,
                            frames=len(times),last_time_s=float(times[-1,0]),
                            increasing_time=True,terminal_index_verified=True,
                            initial_state_frame=initial,
                            terminal_report_steps=terminal['steps']))
    print(run_id, 'read', len([x for x in checked if run_id in x['file']]),flush=True)
plays=[]
for method in ('ppo','bptt','shac','pointcloud','super','ego_planner'):
    row=next(x for x in selection if x['method']==method and x['task']=='racing')
    command=[sys.executable,'scripts/play.py',f"replay=experiments/{row['run_id']}/rollouts",
             'visualization=headless', 'runtime.device=cpu']
    log=ROOT/f'tmp/final-acceptance/play-{method}-final.log'
    with log.open('w') as stream:
        code=subprocess.run(command,cwd=ROOT,stdout=stream,stderr=subprocess.STDOUT,
                            env={**os.environ,'PYTHONPATH':'src','SCIPY_ARRAY_API':'1'}).returncode
    plays.append(dict(method=method,command=command,exit_code=code,log=str(log.relative_to(ROOT))))
    if code:
        raise RuntimeError(f'Play failed for {method}; see {log}')
data=dict(passed=True,readback_count=len(checked),replays=checked,play_commands=plays,
          reader='Python pickle/numpy structural and time readback; headless public play entry')
(OUTPUT/'replay-readback.json').write_text(json.dumps(data,indent=2)+'\n')
print('replay readback',len(checked),'play entries',len(plays),flush=True)
